% Referencias inteligentes "\cref (\nameref)"
\newcommand{\myref}[1]{%
    \cref{#1}%
    \begingroup
    \edef\temp{\getrefbykeydefault{#1}{name}{}}%
    \ifx\temp\empty
        % No hacer nada si \temp está vacío
    \else
        ~(\nameref{#1})% Imprime nombre entre paréntesis en caso de existir
    \fi
    \endgroup
}
