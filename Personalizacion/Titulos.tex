%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%% PARTES %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\renewcommand{\thepart}{\arabic{part}}
% Caja para el número de la parte
\newsavebox{\felinepartnumber}

% Dimensiones del recuadro
\newlength{\partboxwidth}
\newlength{\partboxheight}
\setlength{\partboxwidth}{4.2cm}
\setlength{\partboxheight}{2cm}

% Marcador: "PARTE" vertical + número
\newcommand{\felinepartmarker}{%
    \sbox{\felinepartnumber}{%
        \begingroup
        \setlength{\fboxsep}{4pt}%
        \colorbox{rojoCIMAT}{%
            \makebox[\partboxwidth][c]{%
                \resizebox{!}{5.6cm}{%
                    \normalfont
                    \bfseries
                    \color{white}%
                    \thepart
                }%
            }%
        }%
        \endgroup
    }%
    \makebox[\textwidth][r]{%
        {%
            \normalfont
            \rmfamily
            \mdseries
            \color{black}%
            \rotatebox{90}{%
                \resizebox{%
                    \heightof{\usebox{\felinepartnumber}}%
                    +\depthof{\usebox{\felinepartnumber}}%
                }{!}{%
                    \scshape\so{PARTE}%
                }%
            }%
        }%
        \hspace{0.35em}%
        \raisebox{\depthof{\usebox{\felinepartnumber}}}{%
            \usebox{\felinepartnumber}%
        }%
    }%
}

% Espaciado de las partes
\titlespacing*{\part}
    {0pt}
    {-5em}
    {3em}

% Formato de las partes
\titleformat{\part}[display]
    {\Huge\bfseries\itshape\color{grisCIMAT!80!black}\raggedleft}
    {
    \begin{minipage}[t]{\textwidth}
        \smash{\raisebox{-0.95cm}{\includegraphics[height=1.5cm]{Portada/escudo_cimat.pdf}}} %\hfill
        {\Large\color{rojoCIMAT}\normalfont\bfseries{Centro de Investigación}}
        \vspace{-0.95em}\\
        {\color{rojoCIMAT}\hspace{1.4cm}\rule{\dimexpr\textwidth-10.45cm\relax}{1.3pt}}
        \vspace{-0.55em}\\
        {\Large\color{rojoCIMAT}\normalfont\bfseries\hspace{1.4cm}en Matemáticas, A.C.}
    \end{minipage}\\
    \vspace{4.9cm}
    \felinepartmarker
    }
    {1.5em}
    {}
    [\input{Portada/banner}]
